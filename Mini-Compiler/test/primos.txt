integer n
integer d
integer t
t = 0
n = 2

while n < 100
    d = 2

    while d*d<=n && n/d*d != n
        d = d+1
    end

    if d*d > n
        print n +" es primo!"
        t = t+1
    else
        print n +" no"
    end

    n = n+1
end

print ""

if t==25
    print "OK: "+ t +" primos"
else
    print "ERROR: ho hay "+ t +" primos"
end
